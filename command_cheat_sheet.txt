# Enable X-forwarding from container (run outside docker)
xhost +local:docker

# Run zenoh router
zenohd

# Start basic simulation
just l_turtle_2

# Teleoperate turtle
just teleop

# Start web interface for sending actions
just web

# Publish to /cmd_vel
ros2 topic pub /cmd_vel geometry_msgs/msg/Twist "{linear: {x: 0.1, y: 0.0, z: 0.0}, angular: {x: 0.0, y: 0.0, z: 1.0}}"

# Rosbag /cmd_vel topic
ros2 bag record /cmd_vel

# Xacro -> .sdf
xacro src/dynobot/dynobot_description/urdf/dynobot.xacro -o dynobot.urdf

gz sdf -p dynobot.urdf > dynobot.sdf
